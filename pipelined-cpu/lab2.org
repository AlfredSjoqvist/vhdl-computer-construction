** 3.1
157 clock cycles
#+begin_src asm
          movhi	r1,	0		;
          movhi	r2,	0		;
          nop				; R2 read 2 below
          addi	r1,	r1,	8	; Loop counter = 8
          addi	r2,	r2,	20	; Pointer = 20
          nop				;
          nop				;
  LOOP:					;
          lw	r3,	0(r2)		; Fetch first number
          lw	r4,	20(r2)		; Fetch second number
          nop				;
          nop				;
          muls	r5,	r4,	r3	; Multiply them
          nop				;
          nop				;
          add	r6,	r6,	r5 	; Accumulate to result
          addi	r2,	r2,	4	; Pointer++
          addi	r1,	r1,	-1	; Loop counter --
          nop				;
          nop				;
          sfne	r0,	r1		; Set flag if loop counter != 0
          nop				;
          nop				;
          bf	LOOP			; Continue?
          nop				;
          nop				;
          sw	0(r0),	r6		; Save the result to memory
          trap	0			; Halt execution
          nop				;
          nop				;
#+end_src
